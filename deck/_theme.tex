% The Ineris deck template, carried over from chambers_presentation.pptx:
% colours are the theme's own (ppt/theme/theme2.xml), and the logo is the
% master's image1.jpg. Figures are transparent, so the ground stays white.
\usepackage{helvet}
\usepackage{tikz}
\renewcommand{\familydefault}{\sfdefault}

\definecolor{inerisblue}{HTML}{000091}
\definecolor{inerisgreen}{HTML}{005841}
\definecolor{inerisred}{HTML}{E1000F}
\definecolor{ink}{HTML}{2B2620}
\definecolor{muted}{HTML}{7A7266}
\definecolor{rule}{HTML}{CBC3B3}

\setbeamercolor{structure}{fg=inerisblue}
\setbeamercolor{frametitle}{fg=inerisblue, bg=white}
\setbeamercolor{normal text}{fg=ink, bg=white}
\setbeamercolor{title}{fg=inerisblue}
\setbeamercolor{author}{fg=muted}
\setbeamercolor{institute}{fg=muted}
\setbeamercolor{date}{fg=muted}
\setbeamercolor{itemize item}{fg=inerisblue}
\setbeamercolor{itemize subitem}{fg=muted}
\setbeamercolor{alerted text}{fg=inerisred}

\setbeamerfont{frametitle}{size=\Large, series=\bfseries}
\setbeamerfont{itemize/enumerate body}{size=\large}
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{itemize item}{\textbullet}

% Both marks top-left on every slide, at the master's proportions:
% Republique Francaise at (0.33,0.12) 0.63in, INERIS at (1.04,0.19) 0.83in.
\setbeamertemplate{headline}{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=north west, inner sep=0pt] at ([shift={(0.33in,-0.12in)}]current page.north west)
      {\includegraphics[width=0.63in]{assets/ineris_cover.jpg}};
    \node[anchor=north west, inner sep=0pt] at ([shift={(1.04in,-0.19in)}]current page.north west)
      {\includegraphics[width=0.83in]{assets/ineris_logo.jpg}};
  \end{tikzpicture}}

% Frame title, indented clear of the marks, over the institute's blue rule.
\setbeamertemplate{frametitle}{%
  \vspace{0.75em}\hspace{2.0in}\insertframetitle\par
  \hspace{0.45em}{\color{inerisblue}\rule{0.93\linewidth}{1.2pt}}}

% Footer: the slide number alone; the marks are in the header.
\setbeamertemplate{footline}{%
  \hfill{\color{muted}\tiny\insertframenumber}\hspace{0.6em}\vspace{0.35em}}

% Title slide: the institute's cover image, its logo, and nothing else.
\setbeamertemplate{title page}{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=north east, inner sep=0.6cm] at (current page.north east)
      {\includegraphics[height=1.5cm]{assets/ineris_logo.jpg}};
  \end{tikzpicture}%
  \vspace{1.7cm}
  % Held clear of the logo in the top-right corner.
  \begin{minipage}{0.74\linewidth}
    {\color{inerisblue}\Huge\bfseries\inserttitle\par}
  \end{minipage}\par
  \vspace{0.4em}{\color{inerisblue}\rule{0.55\linewidth}{1.2pt}}\par
  \vspace{0.9em}{\large\insertauthor\par}
  \vspace{0.2em}{\color{muted}\insertinstitute\par}
  \vspace{0.6em}{\color{muted}\small\insertdate\par}}
